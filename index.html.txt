<script>
    document.getElementById('customImg').src = "data:image/png;base64,iVBORw0KGgoAAAANSUhEUgAAAAUAAAAFCAYAAACNbyblAAAAHElEQVQI12P4//8/w38GIAXDIBKE0DHxgljNBAAO9TXL0Y4OHwAAAABJRU5ErkJggg==";

    const video = document.getElementById('videoPlayer');
    const videoSrc = 'https://vivatv.juanjuiserver.com:3782/stream/play.m3u8';
    
    if (Hls.isSupported()) {
        const hls = new Hls();
        hls.loadSource(videoSrc);
        hls.attachMedia(video);
    } else if (video.canPlayType('application/vnd.apple.mpegurl')) {
        video.src = videoSrc;
    }

    const audio = document.getElementById('audioPlayer');
    const bars = document.querySelectorAll('.eq-bar');
    const danceBg = document.getElementById('danceBg');
    const visualizerBox = document.getElementById('visualizerBox');
    const charSpans = document.querySelectorAll('.djs-char');
    
    let animInterval = null;
    let hueValue = 0;

    function startDynamicAnimation() {
        if (animInterval) return;
        animInterval = setInterval(() => {
            if (audio.paused) return;
            
            hueValue = (hueValue + 12) % 360;
            let secondaryHue = (hueValue + 120) % 360;
            
            // Animación ecualizador
            bars.forEach((bar, index) => {
                let randomHeight = Math.floor(Math.sin(Date.now() / 120 + index) * 35 + 50);
                bar.style.height = randomHeight + '%';
            });

            // Fondo y borde dinámicos
            danceBg.style.filter = `hue-rotate(${hueValue}deg) brightness(0.55) saturate(220%)`;
            visualizerBox.style.borderColor = `hsl(${hueValue}, 100%, 65%)`;

            // Baile independiente y destellos mágicos por cada letra al ritmo
            charSpans.forEach((char, idx) => {
                let time = Date.now() / 90 + idx * 0.7;
                let yMove = Math.sin(time) * 4.5;
                let xMove = Math.cos(time * 0.8) * 2;
                let scaleVal = 1 + Math.abs(Math.sin(time * 1.2)) * 0.18;
                let rotVal = Math.sin(time * 0.5) * 6;
                
                // Destellos aleatorios individuales de brillo extremo
                let isSparkle = (idx + Math.floor(Date.now() / 150)) % 4 === 0;
                let charColor = isSparkle ? '#ffffff' : `hsl(${(hueValue + idx * 20) % 360}, 100%, 88%)`;
                let sparkleGlow = isSparkle ? '0 0 16px #ffffff, 0 0 30px #00ffff' : `0 0 10px hsl(${hueValue}, 100%, 60%), 0 0 20px hsl(${secondaryHue}, 100%, 50%)`;

                char.style.transform = `translate(${xMove}px, ${yMove}px) scale(${scaleVal}) rotate(${rotVal}deg)`;
                char.style.color = charColor;
                char.style.textShadow = `0 2px 4px rgba(0,0,0,0.9), ${sparkleGlow}`;
            });
        }, 70);
    }

    function stopDynamicAnimation() {
        if (animInterval) {
            clearInterval(animInterval);
            animInterval = null;
        }
    }

    const startPlayback = () => {
        audio.play().catch(e => console.log("Audio autoplay pendiente"));
        startDynamicAnimation();
        window.removeEventListener('touchstart', startPlayback);
        window.removeEventListener('click', startPlayback);
    };

    window.addEventListener('touchstart', startPlayback);
    window.addEventListener('click', startPlayback);
    
    audio.addEventListener('play', () => {
        startDynamicAnimation();
    });

    audio.addEventListener('pause', () => {
        stopDynamicAnimation();
    });
</script>
